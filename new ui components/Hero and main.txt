        tab.classList.add('active', 'bg-[#0d0d0d]', 'border-white/80');
        tab.classList.remove('bg-[#09090b]', 'border-[#27272a]');
        const activeInd = tab.querySelector('.feature-indicator');
        if (activeInd) {
          activeInd.classList.remove('opacity-0', 'scale-y-50');
          activeInd.classList.add('opacity-100', 'scale-y-100');
        }
        const activeTitle = tab.querySelector('.feature-title');
        if (activeTitle) {
          activeTitle.classList.remove('text-zinc-400');
          activeTitle.classList.add('text-white');
        }
        const activeDesc = tab.querySelector('.feature-desc');
        if (activeDesc) {
          activeDesc.classList.remove('text-zinc-500');
          activeDesc.classList.add('text-zinc-400');
        }

        // Switch Stage Panel with smooth crossfade & slide animation
        Object.keys(panels).forEach(key => {
          const panel = panels[key];
          if (panel) {
            if (key === target) {
              panel.classList.remove('hidden');
              panel.classList.add('flex');
              // trigger smooth entry animation
              panel.classList.remove('stage-animate');
              void panel.offsetWidth; // trigger reflow
              panel.classList.add('stage-animate');
            } else {
              panel.classList.add('hidden');
              panel.classList.remove('flex', 'stage-animate');
            }
          }
        });
      });
    });

    // 2. Dynamic Typewriter Placeholder Animation cycling exact requested phrases
    const promptInput = document.getElementById('promptInput');
    const typewriterPhrases = [
      "Build a sleek SaaS landing page...",
      "Design a real-time crypto dashboard...",
      "Create a Stripe e-commerce store..."
    ];
    let phraseIndex = 0;
    let charIndex = 0;
    let isDeleting = false;
    let typewriterDelay = 70;

    function runTypewriter() {
      if (document.activeElement === promptInput || (promptInput && promptInput.value.length > 0)) {
        setTimeout(runTypewriter, 1000);
        return;
      }

      const currentPhrase = typewriterPhrases[phraseIndex];

      if (!isDeleting) {
        promptInput.setAttribute('placeholder', currentPhrase.substring(0, charIndex + 1));
        charIndex++;

        if (charIndex === currentPhrase.length) {
          isDeleting = true;
          typewriterDelay = 2200;
        } else {
          typewriterDelay = 65;
        }
      } else {
        promptInput.setAttribute('placeholder', currentPhrase.substring(0, charIndex - 1));
        charIndex--;

        if (charIndex === 0) {
          isDeleting = false;
          phraseIndex = (phraseIndex + 1) % typewriterPhrases.length;
          typewriterDelay = 400;
        } else {
          typewriterDelay = 30;
        }
      }

      setTimeout(runTypewriter, typewriterDelay);
    }
    runTypewriter();

    // 3. Mode Dropdown Pill
    const modeBtn = document.getElementById('modeSelectBtn');
    const modeMenu = document.getElementById('modeMenu');
    const selectedMode = document.getElementById('selectedMode');
    const modeItems = document.querySelectorAll('.mode-item');

    if (modeBtn && modeMenu) {
      modeBtn.addEventListener('click', (e) => {
        e.stopPropagation();
        modeMenu.classList.toggle('hidden');
        if (specMenu) specMenu.classList.add('hidden');
      });
    }

    modeItems.forEach(item => {
      item.addEventListener('click', () => {
        const val = item.getAttribute('data-value');
        if (selectedMode) selectedMode.textContent = val;
        modeMenu.classList.add('hidden');
      });
    });

    // 4. Spec Dropdown Pill
    const specBtn = document.getElementById('specSelectBtn');
    const specMenu = document.getElementById('specMenu');
    const selectedSpec = document.getElementById('selectedSpec');
    const specItems = document.querySelectorAll('.spec-item');

    if (specBtn && specMenu) {
      specBtn.addEventListener('click', (e) => {
        e.stopPropagation();
        specMenu.classList.toggle('hidden');
        if (modeMenu) modeMenu.classList.add('hidden');
      });
    }

    specItems.forEach(item => {
      item.addEventListener('click', () => {
        const val = item.getAttribute('data-value');
        if (selectedSpec) selectedSpec.textContent = val;
        specMenu.classList.add('hidden');
      });
    });

    document.addEventListener('click', (e) => {
      if (modeMenu && !modeMenu.contains(e.target) && e.target !== modeBtn) {
        modeMenu.classList.add('hidden');
      }
      if (specMenu && !specMenu.contains(e.target) && e.target !== specBtn) {
        specMenu.classList.add('hidden');
      }
    });

    // 5. File Attachment Simulation
    const fileUploadBtn = document.getElementById('fileUploadBtn');
    const specFileInput = document.getElementById('specFileUpload');
    const fileUploadIndicator = document.getElementById('fileUploadIndicator');
    const uploadedFileName = document.getElementById('uploadedFileName');
    const removeFileBtn = document.getElementById('removeFileBtn');

    if (fileUploadBtn && specFileInput) {
      fileUploadBtn.addEventListener('click', () => {
        specFileInput.click();
      });

      specFileInput.addEventListener('change', (e) => {
        if (e.target.files && e.target.files[0]) {
          uploadedFileName.textContent = e.target.files[0].name;
          fileUploadIndicator.classList.remove('hidden');
          fileUploadIndicator.classList.add('inline-flex');
        }
      });
    }

    if (removeFileBtn) {
      removeFileBtn.addEventListener('click', () => {
        specFileInput.value = '';
        fileUploadIndicator.classList.add('hidden');
        fileUploadIndicator.classList.remove('inline-flex');
      });
    }

    // 6. Generate Button Loading State
    const generateBtn = document.getElementById('generateBtn');
    const btnText = document.getElementById('btnText');
    if (generateBtn && btnText) {
      generateBtn.addEventListener('click', () => {
        const original = btnText.textContent;
        btnText.textContent = "Orchestrating Agents...";
        generateBtn.disabled = true;
        setTimeout(() => {
          btnText.textContent = "Plan Generated ✓";
          setTimeout(() => {
            btnText.textContent = original;
            generateBtn.disabled = false;
          }, 1600);
        }, 1100);
      });
    }

    // 7. Architectural Cards Scroll Entrance / Reveal Observer
    const archCards = document.querySelectorAll('.arch-reveal-card');
    if ('IntersectionObserver' in window && archCards.length > 0) {
      const archObserver = new IntersectionObserver((entries, observer) => {
        entries.forEach(entry => {
          if (entry.isIntersecting) {
            const card = entry.target;
            const delay = parseInt(card.getAttribute('data-delay') || '0', 10);
            setTimeout(() => {
              card.classList.add('revealed');
            }, delay);
            observer.unobserve(card);
          }
        });
      }, {
        threshold: 0.15,
        rootMargin: '0px 0px -40px 0px'
      });

      archCards.forEach(card => archObserver.observe(card));
    } else {
      archCards.forEach(card => card.classList.add('revealed'));
    }

    // 8. Ambient Monochrome Wave Canvas
    const canvas = document.getElementById('ambientWaveCanvas');
    if (canvas) {
      const ctx = canvas.getContext('2d');
      let width, height, t = 0;

      function resize() {
        width = canvas.width = window.innerWidth;
        height = canvas.height = window.innerHeight;
      }
      window.addEventListener('resize', resize);
      resize();

      function draw() {
        t += 0.005;
        ctx.clearRect(0, 0, width, height);
        ctx.lineWidth = 1;
        ctx.strokeStyle = 'rgba(255, 255, 255, 0.06)';

        for (let j = 0; j < 3; j++) {
          ctx.beginPath();
          const offset = j * 45;
          for (let x = 0; x < width; x += 30) {
            const y = height * 0.35 + Math.sin(x * 0.003 + t + j) * 40 + Math.cos(x * 0.001 - t) * 25 + offset;
            if (x === 0) ctx.moveTo(x, y);
            else ctx.lineTo(x, y);
          }
          ctx.stroke();
        }
        requestAnimationFrame(draw);
      }
      draw();
    }
  })();
</script>
