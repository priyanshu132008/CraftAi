import Link from "next/link";
import { motion } from "framer-motion";

export default function DarkNavbar({ logoText, navLinks, ctaText }) {

    { name: "Pricing", href: "#" },
    { name: "About", href: "#" }
  ];

  return (
    <motion.nav 
      initial={{ y: -50, opacity: 0 }}
      animate={{ y: 0, opacity: 1 }}

      <div className="text-xl font-bold tracking-widest">
        {logoText || "CRAFT.AI"}
      </div>
      
      <ul className="hidden md:flex space-x-8 text-sm font-medium text-gray-300">
        {links.map((link, index) => (
          <li key={index}>
            <Link href={link.href} className="hover:text-white transition-colors">
              {link.name}
            </Link>
          </li>
        ))}
      </ul>
      
      <button className="px-5 py-2 text-sm font-semibold bg-white text-black rounded-full hover:bg-gray-200 transition-colors">
        {ctaText || "Login"}
      </button>

}
